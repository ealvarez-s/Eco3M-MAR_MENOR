Como instalar:
OPCION 1> Non-interactive (silent) installation, which allows you to define the installation configuration only once and does not require any user input during installation:
Como root:
sudo sh ./[Toolkit Name]Kit_[version].sh -a --silent --eula accept
Como user:
sh ./[Toolkit Name]Kit_[version].sh -a --silent --eula accept

OPCION 2> Interactive mode, which prompts you to select or confirm certain options during the installation process:
Como root:
sudo sh ./[Toolkit Name]Kit_[version].sh -a --cli
Como user:
sh ./[Toolkit Name]Kit_[version].sh -a --cli

### Pnetcdf
wget https://parallel-netcdf.github.io/Release/pnetcdf-1.12.3.tar.gz
tar xf pnetcdf-1.12.3.tar.gz
cd pnetcdf-1.12.3

### (por seguridad) eliminar intentos de compilacion anteriores
make clean
rm -rf config.log
rm -rf autom4te.cache

### Exportar wrappers (opcional), los base compilers son icx en lugar de icc y ifx en lugar de ifort
export FC=mpiifort
export CC=mpiicx
export CXX=mpiicpx
export F77=mpiifort
export F77=mpiifort

## Por seguridad, quitar posibles rutas a conda y forzar de nuevo los links mpi
unset CFLAGS CXXFLAGS FFLAGS LDFLAGS CPATH LIBRARY_PATH LD_LIBRARY_PATH
source ~/intel/oneapi/setvars.sh --force

./configure --prefix=$HOME/local/pnetcdf-1.12.3 CC=mpiicx CXX=mpiicpx FC=mpiifort F77=mpiifort --enable-shared --enable-parallel --enable-fortran
make
make install

export PNETCDF_DIR=$HOME/local/pnetcdf-1.12.3
export PATH=$PNETCDF_DIR/bin:$PATH
export LD_LIBRARY_PATH=$PNETCDF_DIR/lib:$LD_LIBRARY_PATH



## HDF5
wget https://support.hdfgroup.org/ftp/HDF5/releases/hdf5-1.12/hdf5-1.12.1/src/hdf5-1.12.1.tar.gz
tar xf hdf5-1.12.1.tar.gz
cd hdf5-1.12.1
source ~/intel/oneapi/setvars.sh --force

./configure --prefix=$HOME/local/hdf5-1.12.1 CC=mpiicx CXX=mpiicpx FC=mpiifort F77=mpiifort --enable-parallel --enable-shared --enable-fortran CPPFLAGS="-I$HOME/local/zlib-1.2.13/include" LDFLAGS="-L$HOME/local/zlib-1.2.13/lib"
make
make install

export HDF5_DIR=$HOME/local/hdf5-1.12.1
export PATH=$HDF5_DIR/bin:$PATH
export LD_LIBRARY_PATH=$HDF5_DIR/lib:$LD_LIBRARY_PATH


## NetCDF-C
wget https://github.com/Unidata/netcdf-c/archive/refs/tags/v4.9.2.tar.gz -O netcdf-c-4.9.2.tar.gz
tar xf netcdf-c-4.9.2.tar.gz
cd netcdf-c-4.9.2

./configure --prefix=$HOME/local/netcdf-c-4.9.2 --enable-parallel --enable-shared --enable-netcdf-4 --enable-dap --enable-pnetcdf --disable-libxml2 --d\
isable-byterange CC=mpiicx CXX=mpiicpx FC=mpiifort F77=mpiifort CPPFLAGS="-I$HOME/local/pnetcdf-1.12.3/include -I$HOME/local/hdf5-1.12.1/include -I$HOME/local/zlib-1.2.13/include" LDFLAGS="-L$HOME/local/pnetcdf-1.12.3/lib -L$HOME/local/hdf5-1.12.1/lib -L$HOME/local/zlib-1.2.13/lib"
make
make install

export NETCDF_DIR=$HOME/local/netcdf-c-4.9.2
export PATH=$NETCDF_DIR/bin:$PATH
export LD_LIBRARY_PATH=$NETCDF_DIR/lib:$LD_LIBRARY_PATH


## NetCDF-Fortran
wget https://github.com/Unidata/netcdf-fortran/archive/refs/tags/v4.6.1.tar.gz -O netcdf-fortran-4.6.1.tar.gz
tar xf netcdf-fortran-4.6.1.tar.gz
cd netcdf-fortran-4.6.1

./configure --prefix=$HOME/local/netcdf-fortran-4.6.1 --enable-shared CC=mpiicx CXX=mpiicpx FC=mpiifort F77=mpiifort CPPFLAGS="-I$HOME/local/netcdf-c-4.9.2/include -I$HOME/local/hdf5-1.12.3/include -I$HOME/local/pnetcdf-1.12.3/include" LDFLAGS="-L$HOME/local/netcdf-c-4.9.2/lib -L$HOME/local/hdf5-1.12.3/lib -L$HOME/local/pnetcdf-1.12.3/lib -lpnetcdf"
make
make install

export NETCDF_FORTRAN_DIR=$HOME/local/netcdf-fortran-4.6.1
export PATH=$NETCDF_FORTRAN_DIR/bin:$PATH
export LD_LIBRARY_PATH=$NETCDF_FORTRAN_DIR/lib:$LD_LIBRARY_PATH
